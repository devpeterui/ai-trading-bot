    <script>
        function testConnection() {
            fetch('http://localhost:5000/execute', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ symbol: 'EURUSD', action: 'BUY', volume: 0.1 })
            })
            .then(res => res.json())
            .then(data => {
                document.getElementById('status').innerText = 'Connected & Signal Sent!';
                document.getElementById('status').style.color = 'green';
            })
            .catch(err => {
                document.getElementById('status').innerText = 'Bridge Offline (Check Python Console)';
                document.getElementById('status').style.color = 'red';
            });
        }
    </script>
